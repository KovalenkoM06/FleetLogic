# Використовуємо образ з .NET SDK для збірки
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src

# Копіюємо файли проєкту
COPY . .

# Відновлюємо залежності та будуємо
RUN dotnet restore
RUN dotnet publish -c Release -o /app/publish

# Використовуємо легкий образ для запуску
FROM mcr.microsoft.com/dotnet/aspnet:8.0
WORKDIR /app
COPY --from=build /app/publish .

# Налаштовуємо порт, який вимагає Render (8080)
ENV ASPNETCORE_URLS=http://+:8080
EXPOSE 8080

# Запускаємо (перевір, що ім'я DLL правильне)
ENTRYPOINT ["dotnet", "FleetLogic.dll"]
